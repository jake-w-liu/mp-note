\documentclass[tikz,border=3pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}[>=Latex,x=0.95cm,y=4.2cm]
  \draw[->] (-0.15,0) -- (8.45,0) node[right] {$x$};
  \draw[->] (0,-0.02) -- (0,1.18) node[above] {$y$};
  % Boxes of height 1/k over [k,k+1], k=1..7: total area H_7.
  \foreach \k in {1,...,7} {
    \pgfmathsetmacro{\h}{1/\k}
    \fill[blue!10] (\k,0) rectangle (\k+1,\h);
    % Sliver between the box top and the curve 1/x.
    \fill[orange!55] (\k,\h) -- (\k+1,\h) -- plot[domain=\k+1:\k,samples=40] (\x,{1/\x}) -- cycle;
    \draw[blue!70!black,thick] (\k,0) rectangle (\k+1,\h);
  }
  \draw[red!75!black,very thick,domain=1:8,samples=160]
    plot (\x,{1/\x});
  \foreach \x in {0,1,2,...,8} {
    \draw (\x,0.015) -- (\x,-0.015) node[below,font=\scriptsize] {$\x$};
  }
  \foreach \y/\lab in {1/1,0.5/\frac12,0.25/\frac14} {
    \draw (-0.03,\y) -- (0.03,\y) node[left,font=\scriptsize] {$\lab$};
  }
  \node[align=center,font=\small,blue!70!black] at (5.6,0.98)
    {boxes: height $1/k$ over $[k,k+1]$\\total area $H_n$ (here $n=7$)};
  \node[red!75!black,font=\small] at (5.6,0.48) {$y=1/x$, area $\log(n+1)$};
  \node[align=center,font=\small,orange!60!black] at (5.6,0.73)
    {shaded slivers:\\$H_n-\log(n+1)\to\gamma$};
\end{tikzpicture}
\end{document}
